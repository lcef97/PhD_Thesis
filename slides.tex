\begin{frame}
\frametitle{Motivation: territorial disparities and policy relevance}
\begin{itemize}
  \item Italy exhibits persistent territorial divides in infrastructure, services, and outcomes
  \item PNRR / NextGenerationEU and GRINS: demand for granular, data-informed territorial planning
  \item Research quest: quantify \textbf{local} disparities through interpretable models
  \item[$\Rightarrow$] Method of analysis: parametric and uncertainty-informed statistical modelling
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Methodological framework}
\begin{itemize}
  \item Areal (municipality/province) modelling: spatial domain described through adjacency graphs $\Rightarrow$ \textbf{CAR-like} \footcite{CAR, MacNab} and \textbf{convolution} \footcite{BYM, BYM2} latent models
  \item Bayesian inference: interpretable uncertainty, flexible prior setup and model comparison
  \item Computation: integrated Nested Laplace Approximation \footcite{INLA, VBINLA2024} for latent Gaussian models
  \item Recurring issue: 
  \begin{itemize}
  \item Interpretation of hyperpriors $\Rightarrow$ \textbf{Penalised complexity priors} \footcite{PC}
  \item Spatial confounding $\Rightarrow$ \textbf{Spatial+2.0} filtering \footcite{Urdangarin24}
  \end{itemize}
\end{itemize}
\end{frame}

\frametitle{Why SchoolDataIT?}
\begin{itemize}
  \item Policy and research need: harmonised, georeferenced open data regarding school infrastructure and student outcome

  \item Contribution: an R package that retrieves, cleans, harmonises, and maps data at LAU / NUTS-3 level
\end{itemize}
